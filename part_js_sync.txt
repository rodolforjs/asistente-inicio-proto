/* gráfica generativa por ocasión */
function art(k, seed){
  let s = seed*97+13; const r = () => (s = (s*16807)%2147483647)/2147483647;
  const W=200,H=200; let g='';
  if(k==='cumple'||k==='bebe'){ for(let i=0;i<16;i++){ const x=r()*W,y=r()*H,rr=3+r()*9; g+= i%3? `<circle cx="${x}" cy="${y}" r="${rr}" fill="#fff" opacity="${.25+r()*.5}"/>` : `<rect x="${x}" y="${y}" width="${rr*1.6}" height="${rr*.7}" rx="2" fill="#fff" opacity=".6" transform="rotate(${r()*180} ${x} ${y})"/>`; } 
    if(k==='bebe') g+=[70,50,32].map((rad,i)=>`<path d="M${100-rad} 150 a${rad} ${rad} 0 0 1 ${rad*2} 0" fill="none" stroke="#fff" stroke-width="7" opacity="${.5+i*.15}"/>`).join(''); }
  if(k==='navidad'){ for(let i=0;i<14;i++){ const a=i/14*Math.PI*2; g+=`<line x1="150" y1="60" x2="${150+Math.cos(a)*130}" y2="${60+Math.sin(a)*130}" stroke="#fff" stroke-width="2" opacity=".35"/>`; } g+=`<circle cx="150" cy="60" r="16" fill="#fff" opacity=".8"/>`; }
  if(k==='amor'){ g+=`<circle cx="100" cy="95" r="52" fill="#fff" opacity=".45"/><circle cx="145" cy="95" r="52" fill="#ee5a2b" opacity=".4" style="mix-blend-mode:multiply"/>`; }
  if(k==='boda'){ g+=`<circle cx="95" cy="90" r="40" fill="none" stroke="#fff" stroke-width="9" opacity=".85"/><circle cx="135" cy="90" r="40" fill="none" stroke="#fff" stroke-width="9" opacity=".55"/>`; }
  if(k==='oficina'){ for(let i=0;i<5;i++) g+=`<rect x="${90+i*14}" y="${20+i*14}" width="80" height="80" rx="8" fill="none" stroke="#fff" stroke-width="2.5" opacity="${.9-i*.14}" transform="rotate(${i*9} 130 60)"/>`; }
  if(k==='padre'){ for(let i=0;i<9;i++) g+=`<rect x="${20+i*22}" y="${100-(i%4+1)*18}" width="14" height="${(i%4+1)*36}" rx="7" fill="#fff" opacity="${.25+i%3*.2}"/>`; }
  return `<svg viewBox="0 0 ${W} ${H}" preserveAspectRatio="xMidYMid slice" aria-hidden="true">${g}</svg>`;
}
function cardEl(p, extra='', seed=1){
  const c = document.createElement('div'); c.className = `card k-${p.k} ${extra}`;
  c.style.setProperty('--h', p.h+'px');
  c.innerHTML = `<div class="art">${art(p.k,seed)}<span class="mo">${MES[p.m]}</span><span class="d">${p.d}</span></div><div class="meta"><b>${p.n}</b><i>${p.o}</i></div>`;
  return c;
}

let skipped = false;
const wait = s => new Promise(r => gsap.delayedCall(s, r));
const T = tw => new Promise(r => tw.then(r));

function buildField(){
  const f = $('field'); f.querySelectorAll('.col,.mon,.dot').forEach(n=>n.remove());
  for(let m=0;m<12;m++){
    const c = document.createElement('div'); c.className='col'; c.style.left=(m/12*100)+'%'; f.appendChild(c);
    const l = document.createElement('div'); l.className='mon'; l.style.left=(m/12*100+.6)+'%'; l.textContent=MES[m]; f.appendChild(l);
  }
  const dots = []; let s = 7; const rnd = () => (s = (s*16807) % 2147483647) / 2147483647;
  for(let i=0;i<TOTAL-PEOPLE.length;i++){ const m = Math.floor(rnd()*12), d = 1+Math.floor(rnd()*30); dots.push({x:xPct(m,d), y:6+rnd()*88, hit:false}); }
  PEOPLE.forEach((p,i)=>dots.push({x:xPct(p.m,p.d), y:p.y, hit:true, i}));
  dots.sort((a,b)=>a.x-b.x);
  dots.forEach(d=>{
    const el = document.createElement('div'); el.className='dot'+(d.hit?' hit':''); el.style.left=d.x+'%'; el.style.top=d.y+'%';
    if(d.hit){ const lb=document.createElement('span'); lb.className='lb'; lb.textContent=PEOPLE[d.i].n; el.appendChild(lb); d.lb=lb; }
    f.appendChild(el); d.el=el;
  });
  return dots;
}

async function runSync(){
  skipped = false; gsap.globalTimeline.timeScale(REDUCE?30:1);
  const sync = $('sync'); sync.style.display='flex'; sync.scrollTop=0; document.body.style.overflow='hidden';
  $('people').innerHTML=''; $('summary').innerHTML='';
  gsap.set('#enter',{opacity:0,pointerEvents:'none'});
  $('cRead').textContent='0'; $('cFound').textContent='0';
  $('pulse').className='pulse'; $('stxt').textContent='Conectando con Google Calendar';
  gsap.set('#field',{opacity:1,height:'34vh'}); gsap.set('#scan',{opacity:0,left:'0%'});
  gsap.fromTo(sync,{opacity:0},{opacity:1,duration:.5});
  const dots = buildField();

  await wait(.5);
  $('stxt').textContent='Leyendo tus eventos';
  await T(gsap.from(dots.map(d=>d.el),{opacity:0,y:-26,duration:.5,stagger:{amount:1.3,from:'random'},ease:'power2.out'}));

  $('stxt').textContent='Buscando cumpleaños, aniversarios y fechas de regalo';
  gsap.set('#scan',{opacity:1});
  let idx = 0, found = 0; const proxy = {p:0};
  await T(gsap.to(proxy,{p:100,duration:4.2,ease:'power1.inOut',onUpdate(){
    $('scan').style.left = proxy.p+'%';
    $('cRead').textContent = Math.round(proxy.p/100*TOTAL);
    while(idx<dots.length && dots[idx].x<=proxy.p){
      const d = dots[idx++];
      if(d.hit){ found++; $('cFound').textContent = found;
        gsap.to(d.el,{scale:2.6,backgroundColor:'#ee5a2b',boxShadow:'0 0 18px 4px rgba(238,90,43,.7)',duration:.45,ease:'back.out(3)'});
        gsap.to(d.lb,{opacity:1,duration:.4});
      } else gsap.to(d.el,{opacity:.12,duration:.4});
    }
  }}));
  gsap.to('#scan',{opacity:0,duration:.4}); $('cRead').textContent = TOTAL;

  $('stxt').textContent='Armando las fichas';
  const box = $('people'); box.innerHTML='';
  const NC = innerWidth>1000?6:innerWidth>640?3:2;
  const cols = Array.from({length:NC},()=>{ const c=document.createElement('div'); c.className='pcol'; box.appendChild(c); return c; });
  const cards = [];
  PEOPLE.forEach((p,i)=>{ const c = cardEl(p, p.doubt?'doubt':'', i+1);
    if(p.doubt) c.querySelector('.meta').insertAdjacentHTML('beforeend','<div class="ask"><span>¿Es tu hermana?</span><button data-a="si">Sí</button><button data-a="no">No</button></div>');
    cols[i%NC].appendChild(c); cards.push(c); });
  const hits = dots.filter(d=>d.hit);
  gsap.set(cards,{opacity:0});
  hits.forEach(d=>{
    const c = cards[d.i], r = c.getBoundingClientRect(), dr = d.el.getBoundingClientRect();
    gsap.fromTo(c,{x:dr.left+dr.width/2-(r.left+r.width/2), y:dr.top+dr.height/2-(r.top+r.height/2), scale:.12, opacity:0, rotate:gsap.utils.random(-25,25)},
      {x:0,y:0,scale:1,opacity:1,rotate:0,duration:1,ease:'back.out(1.4)',delay:d.i*.11});
    gsap.to(d.lb,{opacity:0,duration:.2,delay:d.i*.11});
  });
  await wait(PEOPLE.length*.11+1);
  gsap.to('#field',{height:84,duration:.9,ease:'power3.inOut'}); gsap.to(hits.map(d=>d.el),{scale:1.5,duration:.6});

  const dc = box.querySelector('.doubt');
  gsap.fromTo(dc,{x:-5},{x:5,duration:.07,repeat:7,yoyo:true,ease:'sine.inOut',onComplete:()=>gsap.set(dc,{x:0})});
  $('pulse').className='pulse done'; $('stxt').textContent='Listo · conectado con Google Calendar';
  $('summary').innerHTML = `Encontré <em>${PEOPLE.length-1} fechas</em> seguras y <em>1</em> por confirmar.`;
  gsap.from('#summary',{opacity:0,y:14,duration:.6});
  gsap.to('#enter',{opacity:1,pointerEvents:'auto',duration:.6});

  dc.querySelectorAll('button').forEach(b=>b.onclick=()=>{
    const si = b.dataset.a==='si';
    dc.classList.remove('doubt'); dc.querySelector('.ask').remove();
    if(si) dc.querySelector('.meta i').textContent += ' · hermana';
    $('summary').innerHTML = `Encontré <em>${PEOPLE.length} fechas</em>. Voy aprendiendo con cada una.`;
    gsap.fromTo(dc,{scale:1.06},{scale:1,duration:.5,ease:'elastic.out(1,.5)'});
  });

  await wait(3.2);
  if($('sync').style.display==='none') return;
  const t = $('toast'); t.textContent='Detecté un bautizo nuevo en tu calendario · 22 ago';
  gsap.fromTo(t,{opacity:0,y:16},{opacity:1,y:0,duration:.5});
  const nc = cardEl(NEW,'live',20); nc.insertAdjacentHTML('afterbegin','<span class="chip">Nueva</span>');
  cols[NC-1].appendChild(nc);
  gsap.from(nc,{y:-140,opacity:0,scale:.8,duration:.9,ease:'bounce.out'});
  await wait(2.6); gsap.to(t,{opacity:0,duration:.5});
}
